## Installation

```sh
pnpm add @ngxi/academicons
```

## Usage

```ts
import { Component } from '@angular/core'
import { AcademiconsConversation } from '@ngxi/academicons'

@Component({
  imports: [AcademiconsConversation],
  template: `<svg academiconsConversation></svg>`
})
export default class {}
```
