## Installation

```sh
pnpm add @ngxi/oui
```

## Usage

```ts
import { Component } from '@angular/core'
import { OuiWordWrapDisabled } from '@ngxi/oui'

@Component({
  imports: [OuiWordWrapDisabled],
  template: `<svg ouiWordWrapDisabled></svg>`
})
export default class {}
```
