- Author: Boxicons

## Installation

```sh
pnpm add @ngxi/bx
```

## Usage

```ts
import { Component } from '@angular/core'
import { BxHeartCircle } from '@ngxi/bx'

@Component({
  imports: [BxHeartCircle],
  template: `<svg bxHeartCircle></svg>`
})
export default class {}
```
