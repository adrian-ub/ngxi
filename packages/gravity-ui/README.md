## Installation

```sh
pnpm add @ngxi/gravity-ui
```

## Usage

```ts
import { Component } from '@angular/core'
import { GravityUiMagnifier } from '@ngxi/gravity-ui'

@Component({
  imports: [GravityUiMagnifier],
  template: `<svg gravityUiMagnifier></svg>`
})
export default class {}
```
