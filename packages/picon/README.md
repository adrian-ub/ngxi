## Installation

```sh
pnpm add @ngxi/picon
```

## Usage

```ts
import { Component } from '@angular/core'
import { PiconBookmark } from '@ngxi/picon'

@Component({
  imports: [PiconBookmark],
  template: `<svg piconBookmark></svg>`
})
export default class {}
```
