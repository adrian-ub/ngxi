## Installation

```sh
pnpm add @ngxi/reicon
```

## Usage

```ts
import { Component } from '@angular/core'
import { ReiconRuler4 } from '@ngxi/reicon'

@Component({
  imports: [ReiconRuler4],
  template: `<svg reiconRuler4></svg>`
})
export default class {}
```
