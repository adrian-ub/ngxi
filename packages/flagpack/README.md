## Installation

```sh
pnpm add @ngxi/flagpack
```

## Usage

```ts
import { Component } from '@angular/core'
import { FlagpackCi } from '@ngxi/flagpack'

@Component({
  imports: [FlagpackCi],
  template: `<svg flagpackCi></svg>`
})
export default class {}
```
