## Installation

```sh
pnpm add @ngxi/boxicons
```

## Usage

```ts
import { Component } from '@angular/core'
import { BoxiconsClock2Filled } from '@ngxi/boxicons'

@Component({
  imports: [BoxiconsClock2Filled],
  template: `<svg boxiconsClock2Filled></svg>`
})
export default class {}
```
