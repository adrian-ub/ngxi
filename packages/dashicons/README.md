- Author: WordPress
- Website: https://github.com/WordPress/dashicons


## Installation

```sh
pnpm add @ngxi/dashicons
```

## Usage

```ts
import { Component } from '@angular/core'
import { DashiconsShortcode } from '@ngxi/dashicons'

@Component({
  imports: [DashiconsShortcode],
  template: `<svg dashiconsShortcode></svg>`
})
export default class {}
```
