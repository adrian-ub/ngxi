## Installation

```sh
pnpm add @ngxi/octicon
```

## Usage

```ts
import { Component } from '@angular/core'
import { OcticonAlert24 } from '@ngxi/octicon'

@Component({
  imports: [OcticonAlert24],
  template: `<svg octiconAlert24></svg>`
})
export default class {}
```
