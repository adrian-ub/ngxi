## Installation

```sh
pnpm add @ngxi/heroicons-outline
```

## Usage

```ts
import { Component } from '@angular/core'
import { HeroiconsOutlineColorSwatch } from '@ngxi/heroicons-outline'

@Component({
  imports: [HeroiconsOutlineColorSwatch],
  template: `<svg heroiconsOutlineColorSwatch></svg>`
})
export default class {}
```
