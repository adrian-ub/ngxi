```sh
pnpm add @ngxi/pepicons-pencil
```

## Usage

```ts
import { Component } from '@angular/core'
import { PepiconsPencilBookmark } from '@ngxi/pepicons-pencil'

@Component({
  imports: [PepiconsPencilBookmark],
  template: `<svg pepiconsPencilBookmark></svg>`
})
export default class {}
```
