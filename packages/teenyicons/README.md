## Installation

```sh
pnpm add @ngxi/teenyicons
```

## Usage

```ts
import { Component } from '@angular/core'
import { TeenyiconsFaceIdSolid } from '@ngxi/teenyicons'

@Component({
  imports: [TeenyiconsFaceIdSolid],
  template: `<svg teenyiconsFaceIdSolid></svg>`
})
export default class {}
```
