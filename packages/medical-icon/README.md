## Installation

```sh
pnpm add @ngxi/medical-icon
```

## Usage

```ts
import { Component } from '@angular/core'
import { MedicalIconICareStaffArea } from '@ngxi/medical-icon'

@Component({
  imports: [MedicalIconICareStaffArea],
  template: `<svg medicalIconICareStaffArea></svg>`
})
export default class {}
```
