```sh
pnpm add @ngxi/cryptocurrency-color
```

## Usage

```ts

import { CryptocurrencyColorBtc } from '@ngxi/cryptocurrency-color'

@Component({
  imports: [CryptocurrencyColorBtc],
  template: `<svg cryptocurrencyColorBtc></svg>`
})
export default class {}
```
