## Installation

```sh
pnpm add @ngxi/ginetex
```

## Usage

```ts
import { Component } from '@angular/core'
import { GinetexWashing } from '@ngxi/ginetex'

@Component({
  imports: [GinetexWashing],
  template: `<svg ginetexWashing></svg>`
})
export default class {}
```
