- Name: Rivet Icons

- Icons: 210

- Author: Indiana University
- Website: https://github.com/indiana-university/rivet-icons


## Installation

```sh
pnpm add @ngxi/rivet-icons
```

## Usage

```ts
import { Component } from '@angular/core'
import { RivetIconsLightning } from '@ngxi/rivet-icons'

@Component({
  imports: [RivetIconsLightning],
  template: `<svg rivetIconsLightning></svg>`
})
export default class {}
```
