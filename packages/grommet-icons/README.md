## Installation

```sh
pnpm add @ngxi/grommet-icons
```

## Usage

```ts
import { Component } from '@angular/core'
import { GrommetIconsUserExpert } from '@ngxi/grommet-icons'

@Component({
  imports: [GrommetIconsUserExpert],
  template: `<svg grommetIconsUserExpert></svg>`
})
export default class {}
```
