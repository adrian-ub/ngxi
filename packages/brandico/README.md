## Installation

```sh
pnpm add @ngxi/brandico
```

## Usage

```ts

import { BrandicoVimeo } from '@ngxi/brandico'

@Component({
  imports: [BrandicoVimeo],
  template: `<svg brandicoVimeo></svg>`
})
export default class {}
```
