## Installation

```sh
pnpm add @ngxi/material-icon-theme
```

## Usage

```ts
import { Component } from '@angular/core'
import { MaterialIconThemeZip } from '@ngxi/material-icon-theme'

@Component({
  imports: [MaterialIconThemeZip],
  template: `<svg materialIconThemeZip></svg>`
})
export default class {}
```
