## Installation

```sh
pnpm add @ngxi/qlementine-icons
```

## Usage

```ts
import { Component } from '@angular/core'
import { QlementineIconsWindows16 } from '@ngxi/qlementine-icons'

@Component({
  imports: [QlementineIconsWindows16],
  template: `<svg qlementineIconsWindows16></svg>`
})
export default class {}
```
