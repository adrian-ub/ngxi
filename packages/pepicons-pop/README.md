## Installation

```sh
pnpm add @ngxi/pepicons-pop
```

## Usage

```ts
import { Component } from '@angular/core'
import { PepiconsPopBookmark } from '@ngxi/pepicons-pop'

@Component({
  imports: [PepiconsPopBookmark],
  template: `<svg pepiconsPopBookmark></svg>`
})
export default class {}
```
