- Author: Mozilla
- Website: https://github.com/mozilla/fxemoji


## Installation

```sh
pnpm add @ngxi/fxemoji
```

## Usage

```ts
import { Component } from '@angular/core'
import { FxemojiFoxweary } from '@ngxi/fxemoji'

@Component({
  imports: [FxemojiFoxweary],
  template: `<svg fxemojiFoxweary></svg>`
})
export default class {}
```
