## Installation

```sh
pnpm add @ngxi/dinkie-icons
```

## Usage

```ts
import { Component } from '@angular/core'
import { DinkieIconsLockSmallFilled } from '@ngxi/dinkie-icons'

@Component({
  imports: [DinkieIconsLockSmallFilled],
  template: `<svg dinkieIconsLockSmallFilled></svg>`
})
export default class {}
```
