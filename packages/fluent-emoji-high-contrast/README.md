```sh
pnpm add @ngxi/fluent-emoji-high-contrast
```

## Usage

```ts
import { Component } from '@angular/core'
import { FluentEmojiHighContrastAvocado } from '@ngxi/fluent-emoji-high-contrast'

@Component({
  imports: [FluentEmojiHighContrastAvocado],
  template: `<svg fluentEmojiHighContrastAvocado></svg>`
})
export default class {}
```
