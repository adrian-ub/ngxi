## Installation

```sh
pnpm add @ngxi/mi
```

## Usage

```ts
import { Component } from '@angular/core'
import { MiBarChart } from '@ngxi/mi'

@Component({
  imports: [MiBarChart],
  template: `<svg miBarChart></svg>`
})
export default class {}
```
