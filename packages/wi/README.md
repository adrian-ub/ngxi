## Installation

```sh
pnpm add @ngxi/wi
```

## Usage

```ts
import { Component } from '@angular/core'
import { WiDayHail } from '@ngxi/wi'

@Component({
  imports: [WiDayHail],
  template: `<svg wiDayHail></svg>`
})
export default class {}
```
