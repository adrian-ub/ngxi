## Installation

```sh
pnpm add @ngxi/lets-icons
```

## Usage

```ts
import { Component } from '@angular/core'
import { LetsIconsSearchDuotoneLine } from '@ngxi/lets-icons'

@Component({
  imports: [LetsIconsSearchDuotoneLine],
  template: `<svg letsIconsSearchDuotoneLine></svg>`
})
export default class {}
```
