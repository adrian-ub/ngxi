## Installation

```sh
pnpm add @ngxi/lineicons
```

## Usage

```ts
import { Component } from '@angular/core'
import { LineiconsCheck } from '@ngxi/lineicons'

@Component({
  imports: [LineiconsCheck],
  template: `<svg lineiconsCheck></svg>`
})
export default class {}
```
