## Installation

```sh
pnpm add @ngxi/ri
```

## Usage

```ts

import { RiLock2Line } from '@ngxi/ri'

@Component({
  imports: [RiLock2Line],
  template: `<svg riLock2Line></svg>`
})
export default class {}
```
