## Installation

```sh
pnpm add @ngxi/ion
```

## Usage

```ts
import { Component } from '@angular/core'
import { IonCodeDownloadSharp } from '@ngxi/ion'

@Component({
  imports: [IonCodeDownloadSharp],
  template: `<svg ionCodeDownloadSharp></svg>`
})
export default class {}
```
