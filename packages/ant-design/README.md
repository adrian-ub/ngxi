## Installation

```sh
pnpm add @ngxi/ant-design
```

## Usage

```ts
import { Component } from '@angular/core'
import { AntDesignPushpinFilled } from '@ngxi/ant-design'

@Component({
  imports: [AntDesignPushpinFilled],
  template: `<svg antDesignPushpinFilled></svg>`
})
export default class {}
```
