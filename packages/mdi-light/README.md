## Installation

```sh
pnpm add @ngxi/mdi-light
```

## Usage

```ts
import { Component } from '@angular/core'
import { MdiLightCart } from '@ngxi/mdi-light'

@Component({
  imports: [MdiLightCart],
  template: `<svg mdiLightCart></svg>`
})
export default class {}
```
