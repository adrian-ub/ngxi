## Installation

```sh
pnpm add @ngxi/healthicons
```

## Usage

```ts
import { Component } from '@angular/core'
import { HealthiconsColdChain } from '@ngxi/healthicons'

@Component({
  imports: [HealthiconsColdChain],
  template: `<svg healthiconsColdChain></svg>`
})
export default class {}
```
