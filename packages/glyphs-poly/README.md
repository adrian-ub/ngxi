## Installation

```sh
pnpm add @ngxi/glyphs-poly
```

## Usage

```ts
import { Component } from '@angular/core'
import { GlyphsPolyMouse2 } from '@ngxi/glyphs-poly'

@Component({
  imports: [GlyphsPolyMouse2],
  template: `<svg glyphsPolyMouse2></svg>`
})
export default class {}
```
