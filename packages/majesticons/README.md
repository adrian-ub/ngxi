## Installation

```sh
pnpm add @ngxi/majesticons
```

## Usage

```ts
import { Component } from '@angular/core'
import { MajesticonsChatsLine } from '@ngxi/majesticons'

@Component({
  imports: [MajesticonsChatsLine],
  template: `<svg majesticonsChatsLine></svg>`
})
export default class {}
```
