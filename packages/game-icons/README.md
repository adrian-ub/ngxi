## Installation

```sh
pnpm add @ngxi/game-icons
```

## Usage

```ts
import { Component } from '@angular/core'
import { GameIconsDiamondTrophy } from '@ngxi/game-icons'

@Component({
  imports: [GameIconsDiamondTrophy],
  template: `<svg gameIconsDiamondTrophy></svg>`
})
export default class {}
```
