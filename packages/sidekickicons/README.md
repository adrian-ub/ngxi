## Installation

```sh
pnpm add @ngxi/sidekickicons
```

## Usage

```ts
import { Component } from '@angular/core'
import { SidekickiconsCrown } from '@ngxi/sidekickicons'

@Component({
  imports: [SidekickiconsCrown],
  template: `<svg sidekickiconsCrown></svg>`
})
export default class {}
```
