## Installation

```sh
pnpm add @ngxi/pepicons
```

## Usage

```ts
import { Component } from '@angular/core'
import { PepiconsBookmarkPrint } from '@ngxi/pepicons'

@Component({
  imports: [PepiconsBookmarkPrint],
  template: `<svg pepiconsBookmarkPrint></svg>`
})
export default class {}
```
