## Installation

```sh
pnpm add @ngxi/svg-spinners
```

## Usage

```ts
import { Component } from '@angular/core'
import { SvgSpinnersTadpole } from '@ngxi/svg-spinners'

@Component({
  imports: [SvgSpinnersTadpole],
  template: `<svg svgSpinnersTadpole></svg>`
})
export default class {}
```
