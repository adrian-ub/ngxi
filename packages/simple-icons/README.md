## Installation

```sh
pnpm add @ngxi/simple-icons
```

## Usage

```ts
import { Component } from '@angular/core'
import { SimpleIconsWise } from '@ngxi/simple-icons'

@Component({
  imports: [SimpleIconsWise],
  template: `<svg simpleIconsWise></svg>`
})
export default class {}
```
