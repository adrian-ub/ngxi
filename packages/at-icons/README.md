## Installation

```sh
pnpm add @ngxi/at-icons
```

## Usage

```ts
import { Component } from '@angular/core'
import { AtIconsShield } from '@ngxi/at-icons'

@Component({
  imports: [AtIconsShield],
  template: `<svg atIconsShield></svg>`
})
export default class {}
```
