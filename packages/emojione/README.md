## Installation

```sh
pnpm add @ngxi/emojione
```

## Usage

```ts
import { Component } from '@angular/core'
import { EmojioneAnxiousFaceWithSweat } from '@ngxi/emojione'

@Component({
  imports: [EmojioneAnxiousFaceWithSweat],
```
