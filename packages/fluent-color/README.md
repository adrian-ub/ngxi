## Installation

```sh
pnpm add @ngxi/fluent-color
```

## Usage

```ts
import { Component } from '@angular/core'
import { FluentColorMic20 } from '@ngxi/fluent-color'

@Component({
  imports: [FluentColorMic20],
  template: `<svg fluentColorMic20></svg>`
})
export default class {}
```
